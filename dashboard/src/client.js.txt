const $=id=>document.getElementById(id);
let state=null,busy=false,polling=false,page=0,hasMore=false,nextPoll=0,currentView='overview';
const views={overview:['运行概览','查看账号状态，快速测试与执行。'],plan:['步数计划','设置每日目标，管理自动执行。'],history:['执行记录','查看任务进度与云端核对结果。'],membership:['激活与续期','管理使用期，兑换激活码。']};
function showView(view,focus=false){
 currentView=views[view]?view:'overview';
 document.querySelectorAll('[data-page]').forEach(p=>p.hidden=p.dataset.page!==currentView||!state?.signed_in);
 document.querySelectorAll('.nav-item').forEach(b=>{if(b.dataset.view===currentView)b.setAttribute('aria-current','page');else b.removeAttribute('aria-current');});
 $('view-title').textContent=views[currentView][0];$('view-description').textContent=views[currentView][1];
 if(focus)$('view-title').focus({preventScroll:true});
}
document.querySelectorAll('[data-view]').forEach(b=>b.onclick=()=>showView(b.dataset.view,true));
const when=t=>t?new Date(t*1000).toLocaleString('zh-CN',{timeZone:'Asia/Shanghai',hour12:false}):'—';
const steps=n=>n===null||n===undefined?'未读取':n.toLocaleString()+' 步';
function syncAccess(access=state?.account?.membership){
 if(state?.account&&access)state.account.membership=access;
 const active=!!access?.active;
 for(const id of ['run-now','test-connection'])$(id).disabled=busy||!active;
 document.querySelectorAll('[data-member-action]').forEach(b=>b.disabled=busy||!active);
 if(!access)return;
 $('redeem-button').disabled=busy||access.suspended;
 $('account-state').textContent=!active?'当前使用期无效，任务执行已暂停；设置和历史记录仍保留。':state.account.needs_login?'Zepp 凭据失效，请退出后重新登录。':state.account.enabled?'自动执行已开启，使用期内关闭网页也会继续。':'自动执行已暂停，可手动执行一次。';
 $('membership-state').className='notice '+(access.active?'success':'error');
 $('membership-state').textContent=access.suspended?'账号已被管理员停用，请联系管理员。':active?'使用期有效 · 到期时间：'+when(access.expires_at)+'（北京时间）':access.expires_at?'使用期已于 '+when(access.expires_at)+' 到期，请兑换激活码续期。':'尚未激活，请兑换激活码开通。';
 $('member-reference').textContent=state.account.id;
 const status=access.suspended?'已停用':active?'使用中':access.expires_at?'已到期':'待激活';
 $('access-badge').textContent=status;$('access-badge').className='access-badge '+(active?'is-active':'is-inactive');
 $('overview-access').textContent=status;
 $('overview-expiry').textContent=access.expires_at?'到期 '+when(access.expires_at):'兑换激活码即可开通';
 $('overview-plan').textContent=!active?'不可执行':state.account.needs_login?'需重新登录':state.account.enabled?'已开启':'已暂停';
 $('overview-range').textContent=Number(state.account.min_step).toLocaleString()+' – '+Number(state.account.max_step).toLocaleString()+' 步 / 天';
}
async function licenses(){const d=await api('/api/license');syncAccess(d.membership);const box=$('redemption-list');box.replaceChildren();if(!d.redemptions.length)box.textContent='暂无激活码兑换记录。';for(const r of d.redemptions){const item=element('div',undefined,'run');item.textContent=r.hint+' · '+r.duration_days+' 天 · '+when(r.created_at)+' · 兑换后到期 '+when(r.new_expiry);box.append(item);}}
const verifyLabels={checking:'正在读回核对',not_checked:'尚未核对',matched:'云端步数已达到目标',below_target:'云端步数暂未达到目标，可稍后核对',unavailable:'暂未读回步数，效果尚未核实',readable:'当天步数可读取'};
function notice(text,error=false){$('notice').textContent=text;$('notice').className='notice '+(error?'error':'success');$('notice').hidden=!text;}
async function api(path,data){
 const response=await fetch(path,{method:data===undefined?'GET':'POST',headers:data===undefined?{}:{'Content-Type':'application/json','X-CSRF-Token':state?.csrf||''},body:data===undefined?undefined:JSON.stringify(data),credentials:'same-origin'});
 let result;try{result=await response.json();}catch{throw new Error('网站暂时没有返回有效响应，请刷新后重试。');}
 if(!response.ok)throw new Error(result.error||'请求失败，请重试。');return result;
}
async function action(fn){if(busy)return;busy=true;document.querySelectorAll('button').forEach(b=>b.disabled=true);try{await fn();}catch(e){notice(e.message,true);}finally{busy=false;document.querySelectorAll('button').forEach(b=>b.disabled=false);syncAccess();}}
function resetHistory(){page=0;$('filter-status').value='all';$('filter-day').value='';}
async function load(){
 state=await api('/api/status');$('loading-panel').hidden=true;$('login-panel').hidden=state.signed_in;$('account-panel').hidden=!state.signed_in;$('header-user').hidden=!state.signed_in;$('admin-link').hidden=state.signed_in;document.body.classList.toggle('signed-in',state.signed_in);showView(currentView);
 if(state.signed_in){const a=state.account;$('account-label').textContent=a.label;$('min-step').value=a.min_step;$('max-step').value=a.max_step;$('enabled').checked=a.enabled;$('account-state').textContent=a.needs_login?'登录凭据已失效，请退出后重新登录，再开启计划。':a.enabled?'自动执行已开启，关闭网页也会继续。':'自动执行已暂停，可手动执行一次。';await runs();await licenses();}
 else{$('runs-list').replaceChildren();$('connection-result').textContent='尚未测试';}
}
function element(tag,text,cls){const e=document.createElement(tag);if(text!==undefined)e.textContent=text;if(cls)e.className=cls;return e;}
async function runs(){
 const session=state?.csrf;
 const params=new URLSearchParams({page:String(page),filter:$('filter-status').value,day:$('filter-day').value});
 const data=await api('/api/runs?'+params);if(!state?.signed_in||state.csrf!==session)return;
 syncAccess(data.membership);const box=$('runs-list');const expanded=new Set([...box.querySelectorAll('details[open]')].map(e=>e.dataset.id));box.replaceChildren();
 $('stat-total').textContent=data.stats.total;$('stat-accepted').textContent=data.stats.accepted;$('stat-matched').textContent=data.stats.matched;
 const c=data.check;$('connection-result').textContent=!c?(state.account.membership.active?'尚未测试，点击「重新测试」检查连接。':'使用期有效后可测试账号连接。'):['pending','queued','running'].includes(c.status)?'正在测试连接，请稍候…':(c.status==='success'?'连接测试通过':'连接测试未通过')+' · '+c.message+(c.observed_step!==null?' '+c.day+' 云端：'+steps(c.observed_step):'')+' · '+when(c.finished_at||c.created_at);
 hasMore=data.has_more;$('prev-page').hidden=page===0;$('next-page').hidden=!hasMore;$('page-label').textContent='第 '+(page+1)+' 页';
 const active=data.stats.active||data.runs.some(r=>['pending','queued','running'].includes(r.status));nextPoll=Date.now()+(active?3000:30000);
 $('history-update').textContent='更新于 '+when(Math.floor(Date.now()/1000))+(page===0?(active?' · 执行中每 3 秒刷新':' · 每 30 秒刷新'):' · 翻页期间暂停自动刷新');
 if(!data.runs.length){box.textContent=state.account.membership.active?'没有符合条件的记录。可以测试连接，或执行一次并核对效果。':'没有符合条件的记录。开通有效使用期后可以测试和执行任务。';return;}
 for(const r of data.runs){
  const check=r.kind==='check',verification=r.kind==='verify',diagnostic=check||verification;
  const names={manual:'手动执行',schedule:'自动执行',check:'只读连接测试',verify:'只读效果核对'};
  const states={pending:'等待排队',queued:'排队中',running:'执行中',success:diagnostic?'测试完成':'接口已接收',failed:'失败',unknown:'提交结果待确认',skipped:'已跳过'};
  const row=element('article',undefined,'run-record'),head=element('div',undefined,'record-head');
  head.append(element('div',names[r.kind]+(check?'':' · 目标 '+steps(r.step)),'record-title'),element('span',r.verification==='matched'?'云端达到目标':states[r.status]||r.status,'badge '+(r.verification==='matched'?'good':['failed','unknown'].includes(r.status)?'attention':'')));
  row.append(head,element('div',when(r.created_at)+' · 数据日期 '+r.day,'record-meta'),element('p',r.message+(diagnostic?'':' · '+(verifyLabels[r.verification]||'尚未核对')),'record-result'));
  const detail=element('details');detail.dataset.id=r.id;detail.open=expanded.has(r.id);detail.append(element('summary','查看测试过程与步数对比'));
  const grid=element('div',undefined,'record-details');
  const diff=typeof r.before_step==='number'&&typeof r.observed_step==='number'?((r.observed_step-r.before_step>=0?'+':'')+(r.observed_step-r.before_step)+' 步'):'无法比较';
  for(const [label,value] of [['任务编号',r.id],['开始 / 完成',when(r.started_at)+' / '+when(r.finished_at)],['提交前读数',check||verification?'本次不提交':steps(r.before_step)],['核对时云端读数',steps(r.observed_step)],['读数变化',diagnostic?'本次只查询':diff],['核对结果',verifyLabels[r.verification]||'尚未核对'],['最近核对时间',when(r.checked_at)]]){const cell=element('div');cell.append(element('span',label),element('strong',value));grid.append(cell);}
  if(r.parent_id){const cell=element('div');cell.append(element('span','对应执行记录'),element('strong',r.parent_id));grid.append(cell);}
  detail.append(grid,element('p','读回数值仅反映 Zepp 云端记录，不证明第三方 App 已同步，也不能排除其他设备同时写入。','hint'));
  if(!diagnostic&&['success','unknown'].includes(r.status)){const button=element('button','重新核对（不重新提交）','text-button');button.type='button';button.dataset.memberAction='true';button.disabled=busy||!state.account.membership.active;button.onclick=()=>action(async()=>{await api('/api/verify',{id:r.id});resetHistory();await runs();notice('已安排只读核对，稍后自动更新原记录的核对结果。');});detail.append(button);}
  row.append(detail);box.append(row);
 }
}
