const $ = id => document.getElementById(id);
let state = null, busy = false;
function notice(text, error=false) { $('notice').textContent=text; $('notice').className='notice '+(error?'error':'success'); $('notice').hidden=!text; }
async function api(path, data) {
 const response=await fetch(path,{method:data===undefined?'GET':'POST',headers:data===undefined?{}:{'Content-Type':'application/json','X-CSRF-Token':state?.csrf||''},body:data===undefined?undefined:JSON.stringify(data),credentials:'same-origin'});
 let result; try { result=await response.json(); } catch { throw new Error('网站暂时没有返回有效响应，请刷新后重试。'); }
 if(!response.ok) throw new Error(result.error||'请求失败，请重试。'); return result;
}
async function action(fn) { if(busy)return; busy=true; const buttons=[...document.querySelectorAll('button')]; buttons.forEach(b=>b.disabled=true); try { await fn(); } catch(e) {notice(e.message,true);} finally {busy=false;buttons.forEach(b=>b.disabled=false);} }
async function load() {
 state=await api('/api/status'); $('login-panel').hidden=state.signed_in; $('account-panel').hidden=!state.signed_in; $('history-panel').hidden=!state.signed_in;
 if(state.signed_in) {const a=state.account; $('account-label').textContent=a.label; $('min-step').value=a.min_step; $('max-step').value=a.max_step; $('enabled').checked=a.enabled; $('account-state').textContent=a.needs_login?'登录凭据已失效，请退出后重新登录，再开启计划。':a.enabled?'自动执行已开启，关闭网页也会继续。':'自动执行已暂停，可手动执行一次。'; await runs();}
}
async function runs() {
 const {runs}=await api('/api/runs'); const box=$('runs-list');box.replaceChildren();
 if(!runs.length){box.textContent='还没有执行记录。保存计划后，可点击「立即执行一次」。';return;}
 const labels={pending:'等待排队',queued:'排队中',running:'执行中',success:'成功',failed:'失败',unknown:'结果待确认',skipped:'已跳过'};
 for(const r of runs){const row=document.createElement('div');row.className='run'; const info=document.createElement('div'); const title=document.createElement('strong');title.textContent=(r.kind==='schedule'?'自动':'手动')+' · '+r.step.toLocaleString()+' 步'; const detail=document.createElement('div'); detail.textContent=new Date(r.created_at*1000).toLocaleString('zh-CN',{timeZone:'Asia/Shanghai',hour12:false})+' · '+r.message;info.append(title,detail);const status=document.createElement('strong');status.textContent=labels[r.status]||r.status;status.className=r.status==='success'?'success':['failed','unknown'].includes(r.status)?'failure':'pending';row.append(info,status);box.append(row);}
}
$('toggle-password').onclick=()=>{const p=$('password');p.type=p.type==='password'?'text':'password';$('toggle-password').textContent=p.type==='password'?'显示':'隐藏';};
$('login-form').onsubmit=e=>{e.preventDefault();action(async()=>{const data={account:$('account').value,password:$('password').value,min_step:18000,max_step:25000,consent:$('consent').checked};$('password').value='';notice('正在验证 Zepp Life，请稍候…');await api('/api/login',data);await load();notice('登录成功。请设置并保存自己的计划。');});};
$('settings-form').onsubmit=e=>{e.preventDefault();action(async()=>{await api('/api/settings',{min_step:$('min-step').value,max_step:$('max-step').value,enabled:$('enabled').checked});await load();notice('计划已保存。');});};
$('run-now').onclick=()=>action(async()=>{await api('/api/run',{});await runs();notice('任务已加入队列，请查看执行记录。');});
$('refresh-runs').onclick=()=>action(runs);
$('logout').onclick=()=>action(async()=>{await api('/api/logout',{});await load();notice('已退出登录。已开启的自动计划继续执行。');});
$('delete-account').onclick=()=>action(async()=>{if(!$('confirm-delete').checked)throw new Error('请先勾选删除确认。');await api('/api/delete',{confirm:'DELETE'});await load();notice('本站账号、凭据和记录已删除。');});
load().catch(e=>notice(e.message,true));
setInterval(()=>{if(state?.signed_in&&!busy&&!document.hidden)runs().catch(()=>{});},30000);
