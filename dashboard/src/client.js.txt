function renderSchedule(){
 syncAthlete();

 const now=scheduleRuntime.now? scheduleRuntime.now*1000+(Date.now()-scheduleReceivedAt):Date.now();
 const a=state.account,access=a.membership,next=scheduleRuntime.next;
 const reason=access.suspended?'账号已停用':!access.active||access.expires_at*1000<=now?'请先激活或续期':a.needs_login?'凭据失效，请重新登录':!a.enabled?'自动计划已暂停':!next.at?next.reason:'';
 const active=!reason,stale=Date.now()-scheduleReceivedAt>90000;
 const due=active&&next.at*1000<=now;
 const ticking=active&&!due&&!stale;
 const local=new Date(now+28800000),day=local.toISOString().slice(0,10);
 const nextLocal=next.at?new Date(next.at*1000+28800000):null;
 const nextDay=nextLocal?.toISOString().slice(0,10),nextTime=nextLocal?.toISOString().slice(11,16);
 $('schedule-card').classList.toggle('is-counting',ticking);

 $('schedule-day').textContent=active?(nextDay===day?'今天':'明天'):'';

 });
}

function checkNotice(check){
 const state=check?.state;
 if(state==='completed')return '连接测试已通过，结果与检查时间见「帮助与支持 → 连接排查」。';
 if(state==='cached')return '上次连接测试已通过'+(check.checked_at?'（'+when(check.checked_at)+'）':'')+'，本次展示已有结果。';
 if(state==='running')return '已有连接测试正在执行，已复用该任务。';
 if(state==='queued')return check.reused?'已有连接测试排队中，已复用该任务。':'已安排只读连接测试，不会修改步数。';
 if(state==='pending')return '测试任务已保存，等待队列恢复；无需重复点击。';
 if(state==='inactive')return '使用期有效后可测试连接。';
 return '暂时无法确认测试任务状态，可稍后查看记录或在帮助中检查连接。';
}

const labNames={inspect:'只读核对',bind:'虚拟设备绑定',append:'分钟明细追加'};
const labStates={checking:'检查中 / 结果未返回',submitted:'已记录提交意图，待核对',verified:'已读回验证',read:'只读完成',blocked:'已阻止',unconfirmed:'结果未确认'};
function labEvidence(e){
 const box=element('div',undefined,'record-details');
 const labels={summary:'Zepp 汇总',detail:'分钟明细合计',before_summary:'提交前汇总',before_detail:'提交前明细',expected_summary:'预期汇总',expected_detail:'预期明细'};
 for(const [k,label] of Object.entries(labels))if(Object.hasOwn(e,k)){const cell=element('div');cell.append(element('span',label),element('strong',steps(e[k])));box.append(cell);}
 if(e.minutes){const cell=element('div');cell.append(element('span','追加的分钟（北京时间）'),element('strong',e.minutes.map(m=>String(Math.floor(m/60)).padStart(2,'0')+':'+String(m%60).padStart(2,'0')).join('、')));box.append(cell);}
 return box;
}
function renderLabSnapshot(r){
 const box=$('lab-snapshot');box.replaceChildren();const e=r.evidence;
 box.append(element('p',when(r.finished_at||r.created_at)+' · '+labNames[r.kind]+' · 数据日期 '+r.day,'hint'));
 if(e?.devices){box.append(element('strong','设备 '+e.devices.length+' 个'));for(const d of e.devices)box.append(element('p','设备尾号 '+d.id_suffix+' · '+(d.active===null?'启用状态未知':d.active?'已启用':'未启用')+' · 产品编号 '+(d.product_id??'未知'),'hint'));}
 if(e)box.append(labEvidence(e));
 box.append(element('p',r.message,'hint'));
 if(e?.state)box.append(element('p',({consistent:'汇总与明细一致，仍需在微信核对。',inconsistent:'汇总与明细不一致：不能认定目标已生效。',summary_only:'仅读到汇总，不能确认当天明细。',ambiguous:'当天记录有多条，不能确定同一数据源。',unavailable:'暂时无法确认当天数据。'})[e.state]||'待核对','hint'));
}
async function loadLabHistory(){
 if(!state?.signed_in)return;
 const session=state.csrf,value=await api('/api/sync-lab');if(!state?.signed_in||state.csrf!==session)return;
 labClaims=value;syncAccess();const box=$('lab-history');box.replaceChildren();
 $('lab-snapshot').textContent='尚未读取。打开本页不会自动请求 Zepp。';
 $('lab-bind-status').textContent=value.bind_claimed?'已存在绑定提交记录，请只读核对；不会重复绑定。':'每个账号最多发起一次绑定。';
 $('lab-append-status').textContent=value.append_claimed?'今天已存在追加提交记录，请只读核对。':'次数按北京时间日期计算。';
 if(!value.records.length){box.append(element('p','暂无诊断记录。','hint'));return;}
 const latest=value.records.find(r=>r.evidence);if(latest)renderLabSnapshot(latest);
 for(const r of value.records){const row=element('details',undefined,'lab-record');row.append(element('summary',when(r.created_at)+' · '+labNames[r.kind]+' · '+(labStates[r.status]||'待核对')),element('p',r.message,'hint'));if(r.evidence)row.append(labEvidence(r.evidence));row.append(element('p','任务编号：'+r.id,'hint'));box.append(row);}
}
async function runLab(kind){
 const consent=kind==='inspect'||$('lab-'+kind+'-consent').checked;
 if(!consent)throw new Error('请先勾选本项操作的确认说明。');
 notice(kind==='inspect'?'正在只读核对设备与明细…':'正在执行一次实验操作，请勿重复提交…',false,{duration:0});
 try{const result=await api('/api/sync-lab/'+kind,{confirm:consent});notice(result.message,result.status==='unconfirmed');}
 finally{if(kind!=='inspect')$('lab-'+kind+'-consent').checked=false;await refreshAuthStatus();await loadLabHistory();}
}
for(const kind of ['inspect','bind','append'])$('lab-'+kind).onclick=()=>action(()=>runLab(kind));
for(const kind of ['bind','append'])$('lab-'+kind+'-consent').onchange=()=>syncAccess();
$('lab-refresh').onclick=()=>action(loadLabHistory);

}

function setHistoryMode(mode){
 historyMode=mode==='details'?'details':'daily';
 $('daily-history-panel').hidden=historyMode!=='daily';$('detailed-history-panel').hidden=historyMode!=='details';
 $('history-daily-tab').setAttribute('aria-pressed',String(historyMode==='daily'));$('history-detail-tab').setAttribute('aria-pressed',String(historyMode==='details'));

}
const dayStatus=d=>d.active_count?'仍有任务执行中':d.unknown_count?'有提交待确认':d.latest_status==='failed'?'末次尝试失败':d.latest_status==='skipped'?'末次尝试已跳过':d.accepted_count?'已有接收记录':'暂无成功接收';
async function loadDayAttempts(day,box){
 const session=state?.csrf;box.textContent='正在读取当天提交…';
 try{
  const data=await api('/api/daily-runs?'+new URLSearchParams({detail:day}));
  if(!box.isConnected||!state?.signed_in||session!==state.csrf)return;
  box.replaceChildren();
  const list=element('ol',undefined,'daily-timeline');
  for(const r of data.attempts){
   const item=element('li'),status={success:'Zepp 已接收',unknown:'提交结果待确认',failed:'失败',skipped:'已跳过',running:'执行中',queued:'排队中',pending:'等待排队'}[r.status]||'待确认';
   item.append(element('span',when(r.finished_at||r.created_at),'daily-time'),element('strong',steps(r.step)),element('span',(r.kind==='manual'?'手动':'自动')+' · '+status,'hint'));
   list.append(item);
  }
  box.append(list);if(!data.attempts.length)box.append(element('p','没有计划提交记录。','hint'));
  const button=element('button',data.has_more?'查看完整当天记录 →':'查看当天全部明细（含测试与核对） →','text-button');button.type='button';
  button.onclick=()=>action(async()=>{page=0;$('filter-day').value=day;$('filter-status').value='all';setHistoryMode('details');await runs();});box.append(button);
 }catch(e){delete box.dataset.loaded;if(box.isConnected)box.textContent='读取失败：'+e.message+'。收起后展开可重试。';}
}
async function loadDailyHistory(){
 if(!state?.signed_in)return;
 const request=++dailyRequest,session=state.csrf;
 const data=await api('/api/daily-runs?'+new URLSearchParams({page:dailyPage,from:$('daily-from').value,to:$('daily-to').value}));
 if(request!==dailyRequest||!state?.signed_in||state.csrf!==session)return;
 const box=$('daily-list'),expanded=new Set([...box.querySelectorAll('details[open]')].map(e=>e.dataset.day));box.replaceChildren();
 dailyMore=data.has_more;$('daily-prev').hidden=dailyPage===0;$('daily-next').hidden=!dailyMore;$('daily-page').textContent='第 '+(dailyPage+1)+' 页';
 $('daily-updated').textContent='更新于 '+new Date().toLocaleTimeString('zh-CN',{timeZone:'Asia/Shanghai',hour12:false});
 if(!data.days.length){box.append(element('p','这段时间没有计划提交记录。可调整日期范围，或在“全部明细”查看连接测试。','daily-empty'));return;}
 for(const d of data.days){
  const row=element('details',undefined,'daily-record');row.dataset.day=d.day;
  const head=element('summary',undefined,'daily-row'),date=element('div'),value=element('div'),time=element('div'),status=element('div');
  date.append(element('strong',d.day),element('small',d.day===data.today?'今天 · 尚未结束':'历史记录'));
  value.append(element('small','最后被 Zepp 接收'),element('strong',d.final_step===null?'—':steps(d.final_step),'daily-step'));
  time.append(element('small','接收时间'),element('span',d.final_at?new Date(d.final_at*1000).toLocaleTimeString('zh-CN',{timeZone:'Asia/Shanghai',hour12:false}):'尚无接收记录'));
  status.append(element('span',dayStatus(d),'badge '+(d.unknown_count||d.latest_status==='failed'?'attention':'')),element('small','已接收 '+d.accepted_count+' / '+d.attempts+' 次'));
  head.append(date,value,time,status,element('span','展开','daily-expand'));
  const detail=element('div',undefined,'daily-expanded');row.append(head,detail);
  row.ontoggle=()=>{head.querySelector('.daily-expand').textContent=row.open?'收起':'展开';if(row.open&&!detail.dataset.loaded){detail.dataset.loaded='true';loadDayAttempts(d.day,detail);}};
  box.append(row);if(expanded.has(d.day))row.open=true;
 }
}
