const $ = id => document.getElementById(id);
let state = {}, busy = false, poll;
const bootstrap = new URLSearchParams(location.hash.slice(1)).get('setup');
if (location.hash) history.replaceState(null, '', location.pathname);
function show(id, visible = true) { $(id).hidden = !visible; }
function notice(id, message, kind = '') { $(id).textContent = message; $(id).className = 'notice ' + kind; show(id); }
async function api(path, data) {
  const response = await fetch(path, {method: data === undefined ? 'GET' : 'POST', credentials: 'same-origin', headers: data === undefined ? {} : {'Content-Type': 'application/json', 'X-CSRF-Token': state.csrf || ''}, ...(data === undefined ? {} : {body: JSON.stringify(data)})});
  const result = await response.json(); if (!response.ok) throw new Error(result.message || '请求失败，请稍后重试。'); return result;
}
function step(n) { for (let i = 1; i <= 3; i++) $('step' + i).className = i === n ? 'current' : i < n ? 'done' : ''; }
function ready(summary) {
  state.summary = summary; show('save-panel', !!summary); if (!summary) return;
  $('verified-account').textContent = summary.account;
  $('verified-steps').textContent = `每日 ${summary.min_step.toLocaleString()}–${summary.max_step.toLocaleString()} 步`;
  $('device-note').textContent = summary.device_found ? '已找到绑定设备。' : '未查到绑定设备；登录有效，但微信等平台能否同步仍需实际验证。';
  $('replace-config').checked = false; step(3);
}
async function refreshRuns() {
  try {
    const result = await api('/api/runs'); $('runs-list').replaceChildren();
    if (!result.runs.length) { $('runs-list').textContent = '还没有执行记录'; return; }
    for (const run of result.runs) {
      const row = document.createElement('div'); row.className = 'run';
      const a = document.createElement('a'); a.href = run.url; a.target = '_blank'; a.rel = 'noreferrer';
      a.textContent = new Date(run.created_at).toLocaleString('zh-CN', {timeZone: 'Asia/Shanghai', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', hour12: false});
      const label = document.createElement('strong');
      label.textContent = run.status !== 'completed' ? (run.status === 'in_progress' ? '运行中' : '排队中') : ({success: '成功', failure: '失败', cancelled: '已取消', skipped: '已跳过', timed_out: '超时'}[run.conclusion] || '已结束');
      label.className = run.status === 'completed' ? run.conclusion : 'pending'; row.append(a, label); $('runs-list').append(row);
    }
  } catch (e) { $('runs-list').textContent = e.message; }
}
async function init() {
  try {
    state = await api('/api/status'); $('repository').textContent = state.repository;
    show('global-status', false); show('setup-panel', !state.configured); show('signin-panel', state.configured && !state.signed_in); show('workspace', state.signed_in); show('logout', state.signed_in);
    if (!state.configured) { $('create-app').disabled = !bootstrap; $('setup-help').textContent = bootstrap ? '已载入专属初始化凭据。下一步会跳转 GitHub。' : '请使用部署时提供的专属初始化链接。'; }
    if (state.signed_in) {
      $('github-user').textContent = state.login; $('repo-link').href = 'https://github.com/' + state.repository;
      if (state.install_url) { $('install-link').href = state.install_url; show('install-link'); }
      step(2); ready(state.summary); await refreshRuns();
    }
  } catch (e) { notice('global-status', e.message, 'error'); }
}
$('create-app').addEventListener('click', async () => {
  $('create-app').disabled = true;
  try { const result = await api('/api/setup/start', {token: bootstrap}); const form = document.createElement('form'); form.method = 'POST'; form.action = result.action; const input = document.createElement('input'); input.type = 'hidden'; input.name = 'manifest'; input.value = result.manifest; form.append(input); document.body.append(form); form.submit(); }
  catch (e) { notice('global-status', e.message, 'error'); $('create-app').disabled = false; }
});
$('show-password').addEventListener('click', () => { const showing = $('password').type === 'password'; $('password').type = showing ? 'text' : 'password'; $('show-password').textContent = showing ? '隐藏' : '显示'; $('show-password').setAttribute('aria-label', showing ? '隐藏密码' : '显示密码'); });
$('zepp-form').addEventListener('submit', async e => {
  e.preventDefault(); if (busy) return; busy = true;
  const values = {account: $('account').value, password: $('password').value, min_step: $('min-step').value, max_step: $('max-step').value};
  $('password').value = ''; $('zepp-fields').disabled = true; $('save').disabled = true; ready(null); step(2);
  notice('login-status', '正在验证登录并获取配置，请稍候…');
  try { const result = await api('/api/zepp/login', values); ready(result.summary); notice('login-status', '验证成功。所需配置已生成，10 分钟内可保存到 GitHub。', 'success'); }
  catch (e) { notice('login-status', e.message, 'error'); }
  finally { values.password = ''; busy = false; $('zepp-fields').disabled = false; $('save').disabled = false; }
});
$('save').addEventListener('click', async () => {
  if (busy) return; if (!$('replace-config').checked) { notice('save-status', '请先勾选配置替换说明。', 'error'); return; }
  busy = true; $('save').disabled = true; $('zepp-fields').disabled = true; notice('save-status', '正在加密保存配置并请求运行…');
  try { const result = await api('/api/config/save', {replace_config: true}); notice('save-status', result.message, 'success'); state.summary = null;
    let remaining = 12; clearInterval(poll); poll = setInterval(() => { if (--remaining <= 0) clearInterval(poll); if (!document.hidden) refreshRuns(); }, 10000); await refreshRuns();
  } catch (e) { notice('save-status', e.message, 'error'); $('save').disabled = false; }
  finally { busy = false; $('zepp-fields').disabled = false; }
});
$('refresh').addEventListener('click', refreshRuns);
$('logout').addEventListener('click', async () => { if (busy) return; try { await api('/api/logout', {}); location.replace('/'); } catch (e) { notice('global-status', e.message, 'error'); } });
window.addEventListener('pagehide', () => { $('password').value = ''; clearInterval(poll); });
init();
