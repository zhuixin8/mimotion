const $=id=>document.getElementById(id);
let session=null,busy=false,currentTab='codes',generated='',newKey='';
const pages={codes:0,users:0,audit:0},more={};
const when=t=>t?new Date(t*1000).toLocaleString('zh-CN',{timeZone:'Asia/Shanghai',hour12:false}):'未开通';
const node=(tag,text,cls)=>{const e=document.createElement(tag);if(text!==undefined)e.textContent=text;if(cls)e.className=cls;return e;};
function notice(text,error=false){$('admin-notice').hidden=!text;$('admin-notice').textContent=text;$('admin-notice').className='notice '+(error?'error':'success');}
async function api(path,data){const r=await fetch('/api/admin/'+path,{method:data===undefined?'GET':'POST',credentials:'same-origin',headers:data===undefined?{}:{'Content-Type':'application/json','X-CSRF-Token':session?.csrf||''},body:data===undefined?undefined:JSON.stringify(data)});let body;try{body=await r.json();}catch{throw new Error('后台响应异常，请重试。');}if(!r.ok)throw new Error(body.error||'操作失败');return body;}
async function action(fn){if(busy)return;busy=true;document.querySelectorAll('button').forEach(b=>b.disabled=true);try{await fn();}catch(e){notice(e.message,true);}finally{busy=false;document.querySelectorAll('button').forEach(b=>b.disabled=false);}}
function download(name,text){const url=URL.createObjectURL(new Blob([text],{type:'text/plain;charset=utf-8'}));const a=node('a');a.href=url;a.download=name;document.body.append(a);a.click();a.remove();setTimeout(()=>URL.revokeObjectURL(url),1000);}
async function load(){session=await api('status');$('admin-login').hidden=session.signed_in;$('admin-workspace').hidden=!session.signed_in;$('admin-logout').hidden=!session.signed_in;if(session.signed_in){await showTab(currentTab);}else{$('generated-panel').hidden=true;generated='';$('generated-codes').value='';$('codes-list').replaceChildren();$('users-list').replaceChildren();$('audit-list').replaceChildren();}}
async function overview(){const d=await api('overview');const box=$('admin-stats');box.replaceChildren();for(const [label,value] of [['用户总数',d.users.total],['使用期有效',d.users.active],['停用用户',d.users.suspended],['可兑换激活码',d.codes.available]]){const item=node('div');item.append(node('strong',String(value)),node('span',label));box.append(item);}}
async function showTab(name){await overview();currentTab=name;document.querySelectorAll('.admin-tab').forEach(e=>e.hidden=e.id!=='tab-'+name);document.querySelectorAll('[data-tab]').forEach(e=>e.setAttribute('aria-current',e.dataset.tab===name?'page':'false'));if(name!=='security')await loaders[name]();}
function pagination(name,data){more[name]=data.has_more;$(name+'-prev').hidden=pages[name]===0;$(name+'-next').hidden=!data.has_more;$(name+'-page').textContent='第 '+(pages[name]+1)+' 页';}
async function codes(){const d=await api('codes?'+new URLSearchParams({page:pages.codes,status:$('codes-filter').value}));pagination('codes',d);const box=$('codes-list');box.replaceChildren();if(!d.codes.length)box.textContent='没有符合条件的激活码。';for(const c of d.codes){const row=node('article',undefined,'admin-row');const status=c.redeemed_by?'已兑换':c.disabled?'已停用':c.valid_until&&c.valid_until<=Date.now()/1000?'已过兑换期限':'可兑换';row.append(node('h3',c.hint+' · '+c.duration_days+' 天'),node('p',status+' · '+(c.note||'无备注'),'record-result'),node('p','生成：'+when(c.created_at)+' · 最晚兑换：'+(c.valid_until?when(c.valid_until):'不限'),'hint'));if(c.redeemed_by)row.append(node('p','兑换用户：'+c.redeemed_by+' · '+when(c.redeemed_at),'hint'));const reveal=node('button','显示完整激活码','text-button');reveal.onclick=()=>action(async()=>{const r=await api('codes/reveal',{id:c.id});const value=node('input');value.readOnly=true;value.value=r.code;value.setAttribute('aria-label','完整激活码');row.append(value);reveal.hidden=true;});row.append(reveal);if(!c.redeemed_by&&!c.disabled){const revoke=node('button','停用此码','danger-button');revoke.onclick=()=>action(async()=>{await api('codes/revoke',{id:c.id});await codes();await overview();notice('激活码已停用。');});row.append(revoke);}box.append(row);}}
function field(label,type,value){const wrap=node('label',label);const input=node('input');input.type=type;input.value=value;wrap.append(input);return {wrap,input};}
async function users(){const d=await api('users?'+new URLSearchParams({page:pages.users,search:$('user-search').value}));pagination('users',d);const box=$('users-list');box.replaceChildren();if(!d.users.length)box.textContent='没有符合条件的用户。';for(const u of d.users){const row=node('article',undefined,'admin-row');row.append(node('h3',u.label||'已删除本站资料'),node('p','用户编号：'+u.account_id,'hint'),node('p',(u.suspended?'已停用':u.expires_at>Date.now()/1000?'使用期有效':'待激活 / 已到期')+' · 到期：'+when(u.expires_at),'record-result'));const form=node('form',undefined,'admin-user-form');const days=field('追加天数','number','7');days.input.min=1;days.input.max=3650;const expiry=field('指定到期时间（北京时间）','datetime-local',u.expires_at?new Date(u.expires_at*1000+28800000).toISOString().slice(0,16):'');const reason=field('操作备注','text','');reason.input.maxLength=200;form.append(days.wrap,expiry.wrap,reason.wrap);const update=async(mode)=>{const data={id:u.account_id,revision:u.revision,mode,reason:reason.input.value};if(mode==='extend')data.days=days.input.value;if(mode==='expiry'){if(!expiry.input.value)throw new Error('请指定到期时间。');data.expires_at=Math.floor(Date.parse(expiry.input.value+'+08:00')/1000);}await api('users/update',data);await users();await overview();notice('用户权限已更新。');};for(const [label,mode] of [['追加时长','extend'],['设置到期时间','expiry'],[u.suspended?'恢复用户':'停用用户',u.suspended?'resume':'suspend']]){const b=node('button',label,mode==='suspend'?'danger-button':'text-button');b.type='button';b.onclick=()=>action(()=>update(mode));form.append(b);}row.append(form);box.append(row);}}
const actionNames={login:'后台登录',create_codes:'生成激活码',reveal_code:'查看激活码',revoke_code:'停用激活码',redeem:'用户兑换',user_extend:'追加用户时长',user_expiry:'设置到期时间',user_suspend:'停用用户',user_resume:'恢复用户',rotate_key:'更换管理密钥'};
async function audit(){const d=await api('audit?page='+pages.audit);pagination('audit',d);const box=$('audit-list');box.replaceChildren();if(!d.events.length)box.textContent='暂无操作记录。';for(const e of d.events){const row=node('article',undefined,'admin-row');row.append(node('h3',(actionNames[e.action]||e.action)+' · '+when(e.created_at)),node('p','操作者：'+(e.actor==='admin'?'管理员':'用户')+' · 对象：'+e.target,'hint'));const detail=node('details');detail.append(node('summary','查看操作明细'));const pre=node('pre');try{pre.textContent=JSON.stringify(JSON.parse(e.details),null,2);}catch{pre.textContent='明细不可用';}detail.append(pre);row.append(detail);box.append(row);}}
const loaders={codes,users,audit};
$('admin-login-form').onsubmit=e=>{e.preventDefault();action(async()=>{const key=$('admin-key').value;$('admin-key').value='';await api('login',{key});await load();notice('已登录管理后台。');});};
$('admin-logout').onclick=()=>action(async()=>{await api('logout',{});await load();notice('已退出管理后台。');});
document.querySelectorAll('[data-tab]').forEach(b=>b.onclick=()=>action(()=>showTab(b.dataset.tab)));
$('create-codes-form').onsubmit=e=>{e.preventDefault();action(async()=>{const valid=$('code-valid-until').value;const r=await api('codes/create',{days:$('code-days').value,count:$('code-count').value,note:$('code-note').value,valid_until:valid?Math.floor(Date.parse(valid+'+08:00')/1000):null});generated=r.codes.map(c=>c.code+' | '+c.days+' 天').join('\n');$('generated-codes').value=generated;$('generated-panel').hidden=false;pages.codes=0;await codes();await overview();notice('已生成 '+r.codes.length+' 个激活码。可下载后分发给用户。');});};
$('download-codes').onclick=()=>download('MiMotion-activation-codes.txt',generated);
$('codes-filter').onchange=()=>action(async()=>{pages.codes=0;await codes();});
$('refresh-codes').onclick=()=>action(codes);$('refresh-audit').onclick=()=>action(audit);
$('search-users').onsubmit=e=>{e.preventDefault();action(async()=>{pages.users=0;await users();});};
for(const name of ['codes','users','audit'])for(const direction of ['prev','next'])$(name+'-'+direction).onclick=()=>action(async()=>{if(direction==='prev'&&pages[name]>0)pages[name]--;if(direction==='next'&&more[name])pages[name]++;await loaders[name]();});
$('rotate-key').onclick=()=>action(async()=>{if(!$('rotate-confirm').checked)throw new Error('请先勾选更换密钥确认。');const r=await api('rotate-key',{confirm:true});newKey=r.key;$('new-key').value=newKey;$('new-key-panel').hidden=false;await load();notice('密钥已更换。请保存新密钥，旧密钥已经失效。');});
$('download-key').onclick=()=>download('MiMotion-admin-key.txt',newKey);
load().catch(e=>notice(e.message,true));
